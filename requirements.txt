# ChatVoice - Dependencies
#
# 1) Instala PyTorch con soporte CUDA primero.
#    Para RTX 5090 (Blackwell/SM_120) necesitas CUDA 12.8:
#      pip install --upgrade --force-reinstall torch torchaudio torchvision --index-url https://download.pytorch.org/whl/cu128
#
#    Para GPUs anteriores (hasta RTX 4090 / Ada Lovelace) con CUDA 12.4:
#      pip install torch==2.6.0+cu124 torchaudio==2.6.0+cu124 torchvision==0.21.0+cu124 --index-url https://download.pytorch.org/whl/cu124
#
# 2) Luego instala el resto:
#      pip install -r requirements.txt

# ── Servidor ──────────────────────────────────────────────────────────────────
fastapi==0.135.3
uvicorn[standard]==0.42.0
python-multipart==0.0.22
pydantic==2.12.5

# ── TTS / Audio ───────────────────────────────────────────────────────────────
TTS==0.22.0
# transformers 4.46.x es el punto dulce:
#   - TTS 0.22.0 necesita BeamSearchScorer + GPT2PreTrainedModel (eliminados en 5.x)
#   - Chatterbox necesita MinPLogitsWarper (añadido en 4.43+)
#   - F5-TTS es compatible con 4.x
transformers==4.46.3
pygame==2.6.1
piper-tts>=1.4.0           # Motor TTS ligero (ONNX, sin GPU requerida)

# ── Nuevos motores TTS (opcionales, instalar por separado) ───────────────────
# F5-TTS: calidad superior a XTTSv2, MIT, ~3 GB descarga, requiere 12+ GB VRAM
#   pip install "f5-tts>=1.1.20"
#   (>=1.1.20 acepta pydantic>=2.x; versiones anteriores requerían pydantic<=2.10.6)
#
# Chatterbox: rápido (<200ms), MIT, marca de agua Perth, requiere 4-6 GB VRAM
#   pip install chatterbox-tts --no-deps
#   pip install "transformers==5.2.0"
#   (--no-deps evita que chatterbox baje torch==2.6.0 sobre la build CUDA 12.8)

# ── Compatibilidad gruut / TTS ────────────────────────────────────────────────
# gruut 2.2.3 (dependencia de TTS) requiere estas versiones estrictas.
# numpy 2.x y networkx 3.x rompen gruut en runtime.
numpy>=1.19.0,<2.0.0
networkx>=2.5.0,<3.0.0

# ── Utilidades ────────────────────────────────────────────────────────────────
